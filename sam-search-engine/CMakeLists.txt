cmake_minimum_required(VERSION 3.16)
project(sam_search_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sam_search src/unicode.cpp src/sam.cpp src/index_map.cpp src/ranker.cpp src/engine.cpp)
target_include_directories(sam_search PUBLIC include)
if(MSVC)
 target_compile_options(sam_search PRIVATE /W4 /permissive-)
else()
 target_compile_options(sam_search PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sam_search_cli src/main.cpp)
target_link_libraries(sam_search_cli PRIVATE sam_search)
enable_testing()
add_executable(sam_search_tests tests/sam_tests.cpp)
target_link_libraries(sam_search_tests PRIVATE sam_search)
add_test(NAME sam_search_tests COMMAND sam_search_tests)
